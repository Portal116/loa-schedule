} catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } /** * Attendance 시트 확보 (없으면 생성 + 헤더/서식 세팅) */ function ensureAttendanceSheet(ss) { let sheet = ss.getSheetByName("Attendance"); } /** * 참여 상태 취소 (해당 유저 + 대상일자 행 삭제) */ function cancelAttendanceInSheet(ss, data) { const user = (data.user || '').toString().trim(); const target = (data.target || '').toString().trim(); if (!user || !target) { return { status: "error", message: "유저명과 대상일자가 필요합니다." }; } const sheet = ensureAttendanceSheet(ss); const removed = removeAttendanceRows(sheet, user, target); if (removed > 0) { const nowStr = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss"); sendAttendanceNotification(data, nowStr, true); } return { status: "success", removed: removed }; } /** * 같은 유저(괄호 계정 표기 무시) + 같은 대상일자 행을 모두 삭제. 삭제된 행 수 반환. */ function removeAttendanceRows(sheet, user, target) { const lastRow = sheet.getLastRow(); if (lastRow < 2) return 0; const baseUser = user.split('(')[0].trim(); const values = sheet.getRange(2, 2, lastRow - 1, 3).getValues(); // B:유저명 ~ D:대상 let removed = 0; // 아래에서 위로 지워야 행 번호가 밀리지 않는다 for (let i = values.length - 1; i >= 0; i--) { const rowUser = (values[i][0] || '').toString().split('(')[0].trim(); const rowTarget = (values[i][2] || '').toString().trim(); if (rowUser === baseUser && rowTarget === target) { sheet.deleteRow(i + 2); removed++; } } return removed; } function sendAttendanceNotification(data, timeStr, isCancel) { if (!DISCORD_WEBHOOK_URL || !DISCORD_WEBHOOK_URL.startsWith('http')) return; const fields = [ { name: '유저명', value: (data.user || '-').toString(), inline: true }, { name: '대상 일자', value: (data.target || '-').toString(), inline: true } ]; if (!isCancel) { fields.push({ name: '상태', value: (data.status || '-').toString(), inline: true }); if (data.memo) fields.push({ name: '메모', value: data.memo.toString(), inline: false }); } const payload = { embeds: [{ title: isCancel ? '↩️ [참여 상태 취소]' : '📅 [참여 상태 등록]', color: isCancel ? 9807270 : 15105570, fields: fields, footer: { text: '등록 시간: ' + timeStr } }] }; try { UrlFetchApp.fetch(DISCORD_WEBHOOK_URL, { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true }); } catch (e) { console.error('디스코드 웹훅 전송 실패:', e); } } /** * Suggestions 시트 확보 (없으면 생성) */ function ensureSuggestionSheet(ss) { let sheet = ss.getSheetByName("Suggestions"); if (!sheet) { sheet = ss.insertSheet("Suggestions"); sheet.appendRow(["접수일시", "유저명", "분류", "내용", "처리상태"]); sheet.getRange(1, 1, 1, 5) .setBackground("#f3f3f3") .setFontWeight("bold") .setHorizontalAlignment("center"); sheet.setFrozenRows(1); // 처리상태 드롭다운 const rule = SpreadsheetApp.newDataValidation() .requireValueInList(['접수', '검토중', '완료', '보류'], true) .build(); sheet.getRange(2, 5, 500, 1).setDataValidation(rule); } const widths = [170, 110, 90, 520, 90]; widths.forEach((w, i) => sheet.setColumnWidth(i + 1, w)); return sheet; } /** * 건의사항 접수 */ function submitSuggestionToSheet(ss, data) { const user = (data.user || '').toString().trim(); const content = (data.content || '').toString().trim(); if (!user || !content) { return { status: "error", message: "유저명과 내용은 필수입니다." }; } const sheet = ensureSuggestionSheet(ss); const nowStr = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss"); sheet.appendRow([ nowStr, user, (data.category || '기타').toString().trim(), content, "접수" ]); const lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 1, 1, 5).setBackground(null).setFontWeight("normal").setVerticalAlignment("top"); [1, 2, 3, 5].forEach(c => sheet.getRange(lastRow, c).setHorizontalAlignment("center")); // 내용은 길 수 있으므로 왼쪽 정렬 + 줄바꿈 sheet.getRange(lastRow, 4).setHorizontalAlignment("left").setWrap(true); sendSuggestionNotification(user, data.category, content, nowStr); return { status: "success" }; } /** * 건의사항 삭제. * 본인이 올린 것 + 처리상태가 '접수'인 것만 지운다. * 이미 검토중/완료로 넘어간 건은 삭제되지 않는다. */ function deleteSuggestionInSheet(ss, data) { const user = (data.user || '').toString().trim(); const content = (data.content || '').toString().trim(); if (!user || !content) { return { status: "error", message: "유저명과 내용이 필요합니다." }; } const sheet = ss.getSheetByName("Suggestions"); if (!sheet) return { status: "error", message: "Suggestions 시트가 없습니다." }; const lastRow = sheet.getLastRow(); if (lastRow < 2) return { status: "error", message: "삭제할 항목이 없습니다." }; const baseUser = user.split('(')[0].trim(); const values = sheet.getRange(2, 2, lastRow - 1, 4).getValues(); // B유저명 C분류 D내용 E처리상태 // 아래에서 위로 훑어 가장 최근 항목 1건만 삭제한다 for (let i = values.length - 1; i >= 0; i--) { const rowUser = (values[i][0] || '').toString().split('(')[0].trim(); const rowContent = (values[i][2] || '').toString().trim(); const rowState = (values[i][3] || '접수').toString().trim(); if (rowUser === baseUser && rowContent === content) { if (rowState !== '접수') { return { status: "ignored", message: "이미 처리가 시작된 건의는 삭제할 수 없습니다." }; } sheet.deleteRow(i + 2); return { status: "success", removed: 1 }; } } return { status: "error", message: "해당 건의를 찾지 못했습니다." }; } function sendSuggestionNotification(user, category, content, timeStr) { if (!DISCORD_WEBHOOK_URL || !DISCORD_WEBHOOK_URL.startsWith('http')) return; } } /** * Character_Pool 시트 I~M열의 야생 체크('W') 토글 함수 */ function toggleWildClearInPool(ss, data) { const poolSheet = ss.getSheetByName("Character_Pool"); if (!poolSheet) return { status: "error", message: "Character_Pool 시트를 찾을 수 없습니다." }; const raidColMap = { "final": 9, // I열: 종막 "act4": 10, // J열: 4막 "cathedral": 11, // K열: 성당 "serka": 12, // L열: 세르카 "belga": 13 // M열: 벨가 }; const colIdx = raidColMap[data.raidKey]; if (!colIdx) return { status: "error", message: "유효하지 않은 레이드 키입니다." }; i++) { const rowChar = (names[i][0] || "").toString().trim(); if (rowChar === data.charName) { const rowIdx = i + 2; const targetCell = poolSheet.getRange(rowIdx, colIdx); const currentVal = (targetCell.getValue() || "").toString().trim().toUpperCase(); targetCell.setHorizontalAlignment("center"); return { status: "success", cleared: true }; } } } return { status: "error", message: "해당 캐릭터를 시트에서 찾지 못했습니다." }; } /** * 슬롯 덮어쓰기 */ function updateExactSlotInSheet(sheet, data) { const values = sheet.getDataRange().getValues(); const oldCharName = (data.oldCharName || '').trim(); const targetRow = parseInt(data.sheetRow, 10); const party = parseInt(data.party, 10); const charCol = (party === 2) ? 6 : 1; // B열(1파티) / G열(2파티) // 1순위: 프론트가 보낸 행 번호가 맞는지 확인하고 그대로 쓴다. if (targetRow && targetRow <= values.length) { const checkChar = (values[targetRow - 1][charCol] || '').toString().trim();
